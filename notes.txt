Quick command reference (run from repo root).

# Install / build
uv sync                                # installs deps and builds the cpp extension

# Self-play (writes artifacts/selfplay_buffer.bin, watches artifacts/engines/.iteration)
uv run python -m src.self_play --boards 128 --threads 1

# Train (reads artifacts/buffers/buffer.bin, saves to artifacts/checkpoints/, ships engines via scripts/build_trt.sh)
uv run python -m src.train --config src/train/config.yaml

# Head-to-head evaluation
uv run python -m src.eval.brawls \
  --num_games 100 \
  --model1 small --model2 small \
  --temp 0.0 --temp2 0.0 \
  --ckpt1 artifacts/checkpoints/<a>.pt \
  --ckpt2 artifacts/checkpoints/<b>.pt

# Manually convert a checkpoint to a TensorRT engine
scripts/build_trt.sh artifacts/checkpoints/step_00043900.pt 43900

# Tests
uv run pytest                          # full
uv run pytest -m "not gpu and not trt" # CPU-only

# Prune old checkpoints/engines (default keeps last 5; dry-run by default)
uv run python scripts/prune_artifacts.py            # dry-run
uv run python scripts/prune_artifacts.py --apply
