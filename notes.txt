Quick command reference (run from repo root).

# Install / build
uv sync                                # installs deps and builds the cpp extension

# Start (or resume) the full training loop
uv run python -m src.run               # spawns self-play + trainer together
uv run python -m src.run --boards 256  # tune self-play parallelism
uv run python -m src.run --no-train      # self-play only
uv run python -m src.run --no-selfplay   # trainer only (replay an existing buffer)

# Watch per-process logs
tail -f artifacts/runs/selfplay.log
tail -f artifacts/runs/training.log

# Head-to-head evaluation between two checkpoints
uv run python -m src.eval.brawls \
  --num_games 100 \
  --model1 small --model2 small \
  --temp 0.0 --temp2 0.0 \
  --ckpt1 artifacts/checkpoints/<a>.pt \
  --ckpt2 artifacts/checkpoints/<b>.pt

# Manually convert a checkpoint to a TensorRT engine
scripts/build_trt.sh artifacts/checkpoints/step_00043900.pt 43900

# Tests
uv run pytest                          # full
uv run pytest -m "not gpu and not trt" # CPU-only

# Prune old checkpoints/engines (default keeps last 5; dry-run by default)
uv run python scripts/prune_artifacts.py
uv run python scripts/prune_artifacts.py --apply
